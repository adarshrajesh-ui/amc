    while(root){
        if(root-&gt;val==val)return true;

unsigned f(unsigned n, unsigned b, unsigned p[]) {
    unsigned rb=0, maxrb=0;
    for(int i=0,j=0;i&lt;n;i++){
        rb+=p[i];
        while(j&lt;=i&amp;&amp;rb&gt;b) rb-=p[j++];
        if(maxrb&lt;rb) maxrb=rb;
    }
    return maxrb;
}
